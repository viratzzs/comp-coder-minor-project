from collections import defaultdict

def main():

        total_tri = n * (n-1) * (n-2) // 6
        colinear = 0

        y0 = 0
        y1 = 1
        count_y0 = 0

                count_y0 += 1

                count_y1 += 1
        colinear += count_y0 * (count_y0 - 1) * (count_y0 - 2) // 6 if count_y0 >= 3 else 0
