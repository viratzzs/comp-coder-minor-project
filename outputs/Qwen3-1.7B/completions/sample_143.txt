import bisect

def main():
    import sys
    input = sys.stdin.read().split()
    ptr = 0
    t = int(input[ptr])
    ptr += 1
    for _ in range(t):
        n, k = int(input[ptr]), int(input[ptr+1])

        freq = Counter(a)
        unique = sorted(freq.keys())
        s = unique
        # Compute prefix sums
        prefix = [0] * (len(s) + 1)
        for i in range(1, len(s) + 1):
            prefix[i] = prefix[i-1] + freq[s[i-1]]
        max_sum = 0
