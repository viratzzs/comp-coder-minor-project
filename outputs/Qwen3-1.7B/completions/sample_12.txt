        found = False
        for i in range(1, n):
            new_possible = set()
            for val in possible:
                new_add = val + a[i]
                if new_add <= m:
                    new_possible.add(new_add)
                new_mul = val * a[i]
                if new_mul <= m:
                    new_possible.add(new_mul)
            possible = new_possible
            if m in possible:
                found = True
                break
        if found:
            results.append("YES")
        else:
            results.append("NO")
    print('\n'.join(results))

if __name__ == "__main__":
    main()
