def main():
    input = sys.stdin.read
    data = input().split()
    idx = 0
    t = int(data[idx])
    idx += 1
    results = []
    for _ in range(t):
        n, x = int(data[idx]), int(data[idx+1])

        a = list(map(int, data[idx:idx+n]))

        residues = set()

        # Find the smallest non-negative integer not in residues

        while m in residues:
