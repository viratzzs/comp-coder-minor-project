    while b:
        a, b = b, a % b

    for i in range(1, n + 1):
        prefix[i] = compute_gcd(prefix[i - 1], a[i - 1])
    return prefix

    suffix = [0] * (n + 1)

    return suffix

    i = 1

            divisors.add(i)
            divisors.add(x // i)
        i += 1
    return divisors
