- Initialize a queue with nodes with in-degree zero.

from collections import deque

def topological_sort(n, adj):
    in_degree = [0] * (n+1)
    for u in range(1, n+1):
        for v in adj[u]:
            in_degree[v] += 1
    queue = deque()
    for u in range(1, n+1):
        if in_degree[u] == 0:
            queue.append(u)
    top_order = []
    while queue:
        u = queue.popleft()
        top_order.append(u)
        for v in adj[u]:
            in_degree[v] -= 1
            if in_degree[v] == 0:
                queue.append(v)
    return top_order
