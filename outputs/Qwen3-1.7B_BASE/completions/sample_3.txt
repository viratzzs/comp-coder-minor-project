2. **Key Insight**: The maximum possible score for each operation is determined by the maximum distance between any two points on the same horizontal line. This is because the area of the triangle formed by two points on one line and one point on the other line is maximized when the points are as far apart as possible.
3. **Optimal Strategy**: For each number of operations from 1 to the maximum possible number of operations, we determine the optimal distribution of points between the two lines to maximize the score. The maximum number of operations is determined by the total number of points divided by 3, as each operation uses exactly three points.

### Solution Code

import sys
import bisect

def main():
    input = sys.stdin.read
    data = input().split()
    idx = 0
    t = int(data[idx])
    idx += 1
    results = []
    
    for _ in range(t):
        n = int(data[idx])
        m = int(data[idx + 1])
        idx += 2
        a = list(map(int, data[idx:idx + n]))
        idx += n
        b = list(map(int, data[idx:idx + m]))
        idx += m
        
        a.sort()
        b.sort()
        
        max_a = a[-1] - a[0]
        max_b = b[-1] - b[0]
        
        k_max = (n + m) // 3
        
        f = []
        for k in range(1, k_max + 1):
            if max_a >= max_b:
                x = min(k, n)
                y = max(k - x, 0)
                f.append(x * max_a + y * max_b)
            else:
                y = min(k, m)
                x = max(k - y, 0)
                f.append(x * max_b + y * max_a)
        
        results.append(f)
    
    print(len(results))
    if len(results) > 0:
        for res in results:
            print(' '.join(map(str, res)))

if __name__ == "__main__":
    main()
