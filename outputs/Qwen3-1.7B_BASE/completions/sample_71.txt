def main():
    input = sys.stdin.read

    idx = 0

    n = int(data[idx])

        prefix_b[i] = prefix_b[i - 1] + b[i - 1]

    g = [0] * (n + 1)

        g[i] = prefix_a[i] + b[i - 1]

    max_prefix = [0] * (n + 2)
    max_suffix = [0] * (n + 2)

    max_g_r_plus_b_r = -float('inf')

    for i in range(1, n + 1):
        if i == 1:
            current_max_g_r_plus_b_r = g[i] + b[i - 1]
