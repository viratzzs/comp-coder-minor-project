        for __ in range(q):
            l = int(data[idx])
            r = int(data[idx+1])
            idx += 2
            total = r - l + 1
            full_cycles = total // n
            remainder = total % n
            sum_a = prefix[n]
            if remainder > 0:
                sum_part = prefix[remainder]
            else:
                sum_part = 0
            results.append(str(full_cycles * sum_a + sum_part))
    print('\n'.join(results))

if __name__ == "__main__":
    main()
