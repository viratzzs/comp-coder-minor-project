def build_sparse_table(a, n):
    log_table = [0] * (n + 1)
    for i in range(2, n + 1):
        log_table[i] = log_table[i // 2] + 1
    st = [[0] * n for _ in range(log_table[n] + 1)]

def build_sparse_table(a, n):
    log_table = [0] * (n + 1)
    for i in range(2, n + 1):
        log_table[i] = log_table[i // 2] + 1
    st = [[0] * n for _ in range(log_table[n] + 1)]
    for i in range(n):
        st[0][i] = a[i]
    for k in range(1, log_table[n] + 1):
        for i in range(n - (1 << k) + 1):
            st[k][i] = max(st[k-1][i], st[k-1][i + (1 << (k-1))])
    return st, log_table

Then, the query function:

def query_max(st, log_table, l, r):
    if l > r:

    k = log_table[r - l + 1]
    return max(st[k][l], st[k][r - (1 << k) + 1])

To solve this problem, we need to count the number of good subarrays in a given array. A subarray is considered good if, when sorted, its two medians (the middle elements when the length is even or odd) are equal. 

### Approach
1. **Understanding Good Subarrays**: A subarray is good if the two middle elements (when sorted) are equal. This means that in the sorted subarray, there must be at least two equal elements that are adjacent in the sorted order.
2. **Efficient Range Queries**: We use a sparse table for range maximum queries to efficiently check if the elements between any two indices are all less than or equal to a given value.
3. **Handling Consecutive Pairs**: For each unique value in the array, we consider consecutive pairs of indices where the value appears. If these indices are consecutive in the original array, we directly compute the contribution of these pairs.
4. **Binary Search for Non-Consecutive Pairs**: For non-consecutive pairs, we use binary search to find the farthest index where the elements between the current and farthest index are all less than or equal to the current value.

### Solution Code
```python
import sys
import bisect
from collections import defaultdict

def build_sparse_table(a, n):
    log_table = [0] * (n + 1)
    for i in range(2, n + 1):
        log_table[i] = log_table[i // 2] + 1
    st = [[0] * n for _ in range(log_table[n] + 1)]
    for i in range(n):
        st[0][i] = a[i]
    for k in range(1, log_table[n] + 1):
        for i in range(n - (1 << k) + 1):
            st[k][i] = max(st[k-1][i], st[k-1][i + (1 << (k-1))])
    return st, log_table

def query_max(st, log_table, l, r):
    if l > r:
        return -1
    k = log_table[r - l + 1]
    return max(st[k][l], st[k][r - (1 << k) + 1])

def main():
    input = sys.stdin.read
    data = input().split()
    idx = 0
    t = int(data[idx])
    idx += 1
    results = []
    for _ in range(t):
        n = int(data[idx])
        idx += 1
        a = list(map(int, data[idx:idx + n]))
        idx += n
        max_n = n
        st, log_table = build_sparse_table(a, max_n)
        freq = defaultdict(list)
        for i in range(n):
            freq[a[i]].append(i)
        answer = 0
        for x in freq:
            S = freq[x]
            k = len(S)
            for i in range(k):
                j = S[i] + 1
                if j in freq[x]:
                    answer += (S[i] + 1) * (n - j)
                else:
                    low = S[i] + 1
                    high = n
                    res = S[i] + 1
                    while low <= high:
                        mid = (low + high) // 2
                        if mid > n:
                            break
                        l = S[i] + 1
                        r = mid - 1
                        if r < l:
                            break
                        current_max = query_max(st, log_table, l, r)
                        if current_max <= x:
                            res = mid
                            low = mid + 1
                        else:
                            high = mid - 1
                    if res > S[i] + 1:
                        answer += (S[i] + 1) * (n - res)
        results.append(answer)
    sys.stdout.write("\n".join(map(str, results)) + "\n")

if __name__ == "__main__":
    main()
```
